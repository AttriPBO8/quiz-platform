/**
 * Authentication Service Module
 * Handles OTP generation, verification, and user authentication
 * Implements rate limiting and retry mechanisms
 */

import { Injectable, BadRequestException, UnauthorizedException } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { User } from '../user/user.entity';
import { RedisService } from '../../common/redis/redis.service';
import { TwilioService } from '../../common/twilio/twilio.service';
import { LoggerService } from '../../common/logger/logger.service';

interface OTPRequest {
  phoneNumber: string;
  countryCode: string;
}

interface OTPVerification {
  phoneNumber: string;
  otp: string;
  fullName?: string;
}

interface LoginResponse {
  accessToken: string;
  refreshToken: string;
  user: {
    id: string;
    phoneNumber: string;
    fullName: string;
    deviceId?: string;
    role: string;
  };
}

@Injectable()
export class AuthService {
  private readonly OTP_EXPIRY = 300; // 5 minutes in seconds
  private readonly OTP_MAX_ATTEMPTS = 3;
  private readonly OTP_RATE_LIMIT = 5; // 5 OTP requests per hour

  constructor(
    @InjectRepository(User)
    private userRepository: Repository<User>,
    private redisService: RedisService,
    private twilioService: TwilioService,
    private logger: LoggerService
  ) {}

  /**
   * Generate and send OTP to user's phone
   * @param request - Contains phone number and country code
   * @returns Success message with OTP expiry info
   */
  async generateOTP(request: OTPRequest): Promise<{ message: string; expiresIn: number }> {
    try {
      const { phoneNumber, countryCode } = request;
      const fullPhoneNumber = `${countryCode}${phoneNumber}`;

      // Check rate limiting
      await this.checkRateLimit(fullPhoneNumber);

      // Generate 6-digit OTP
      const otp = this.generateSecureOTP();
      
      // Store OTP in Redis with expiry
      const key = `otp:${fullPhoneNumber}`;
      await this.redisService.set(key, otp, this.OTP_EXPIRY);

      // Store attempt count
      const attemptKey = `otp:attempts:${fullPhoneNumber}`;
      await this.redisService.increment(attemptKey);
      await this.redisService.expire(attemptKey, this.OTP_EXPIRY);

      // Send OTP via SMS
      await this.twilioService.sendSMS(
        fullPhoneNumber,
        `Your OTP for Quiz Platform is: ${otp}. Valid for 5 minutes.`
      );

      this.logger.log(`OTP sent to ${fullPhoneNumber}`);

      return {
        message: 'OTP sent successfully',
        expiresIn: this.OTP_EXPIRY
      };
    } catch (error) {
      this.logger.error(`OTP generation failed: ${error.message}`);
      throw new BadRequestException('Failed to send OTP. Please try again.');
    }
  }

  /**
   * Verify OTP and authenticate user
   * @param verification - OTP verification data
   * @returns Authentication tokens and user data
   */
  async verifyOTP(verification: OTPVerification): Promise<LoginResponse> {
    try {
      const { phoneNumber, otp, fullName } = verification;
      const fullPhoneNumber = `+${phoneNumber}`;

      // Validate OTP from Redis
      const key = `otp:${fullPhoneNumber}`;
      const storedOTP = await this.redisService.get(key);

      if (!storedOTP || storedOTP !== otp) {
        throw new UnauthorizedException('Invalid or expired OTP');
      }

      // Check attempt limit
      const attemptKey = `otp:attempts:${fullPhoneNumber}`;
      const attempts = await this.redisService.get(attemptKey);
      if (parseInt(attempts) > this.OTP_MAX_ATTEMPTS) {
        throw new UnauthorizedException('Maximum OTP attempts exceeded');
      }

      // Find or create user
      let user = await this.userRepository.findOne({
        where: { phoneNumber: fullPhoneNumber }
      });

      if (!user) {
        // Create new user if not exists
        user = this.userRepository.create({
          phoneNumber: fullPhoneNumber,
          fullName: fullName || `User_${phoneNumber.substring(phoneNumber.length - 4)}`,
          isActive: true,
          role: 'user'
        });
        await this.userRepository.save(user);
      } else if (fullName) {
        // Update user name if provided
        user.fullName = fullName;
        await this.userRepository.save(user);
      }

      // Generate JWT tokens
      const accessToken = this.generateAccessToken(user);
      const refreshToken = this.generateRefreshToken(user);

      // Clean up OTP data
      await this.redisService.del(key);
      await this.redisService.del(attemptKey);

      this.logger.log(`User logged in: ${user.id}`);

      return {
        accessToken,
        refreshToken,
        user: {
          id: user.id,
          phoneNumber: user.phoneNumber,
          fullName: user.fullName,
          deviceId: undefined, // Will be set by client
          role: user.role
        }
      };
    } catch (error) {
      this.logger.error(`OTP verification failed: ${error.message}`);
      throw new UnauthorizedException('OTP verification failed');
    }
  }

  /**
   * Generate secure 6-digit OTP
   * @returns 6-digit OTP string
   */
  private generateSecureOTP(): string {
    return Math.floor(100000 + Math.random() * 900000).toString();
  }

  /**
   * Check rate limiting for OTP requests
   * @param phoneNumber - User's phone number
   * @throws BadRequestException if rate limit exceeded
   */
  private async checkRateLimit(phoneNumber: string): Promise<void> {
    const rateKey = `otp:rate:${phoneNumber}`;
    const currentCount = await this.redisService.get(rateKey);
    
    if (currentCount && parseInt(currentCount) >= this.OTP_RATE_LIMIT) {
      throw new BadRequestException(
        'Too many OTP requests. Please try again after an hour.'
      );
    }

    // Increment rate counter
    await this.redisService.increment(rateKey);
    await this.redisService.expire(rateKey, 3600); // 1 hour expiry
  }

  /**
   * Generate JWT access token
   * @param user - User entity
   * @returns JWT access token
   */
  private generateAccessToken(user: User): string {
    // Implementation using JWT library
    return 'access_token_placeholder';
  }

  /**
   * Generate JWT refresh token
   * @param user - User entity
   * @returns JWT refresh token
   */
  private generateRefreshToken(user: User): string {
    // Implementation using JWT library
    return 'refresh_token_placeholder';
  }
}
